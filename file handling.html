#include <stdio.h>
#include <conio.h>
#include <dos.h>

void main()
{
FILE *f1;
char fname[20],ch;
clrscr();

printf("file name: ");
scanf("%s",fname);

f1=fopen(fname,"w");

do
{
printf("Enter character: ");
scanf(" %c",&ch);

if(ch>='a' && ch<='z')
ch=ch-32;

fputc(ch,f1);

printf("continue(Y/N)? ");
scanf(" %c",&ch);
}
while(ch=='y' || ch=='Y');

fclose(f1);

delay(5000);
}

{

int x;
clrscr();

printf("file name= ");
scanf("%s",fname);

printf("data= ");
scanf("%s",data);

f1=fopen(fname,"w");
fprintf(f1,"%s",data);
fclose(f1);

printf("2nd file name= ");
scanf("%s",fname2);

f1=fopen(fname,"r");
f2=fopen(fname2,"w");

while((x=fgetc(f1))!=-1)
{
fputc(x,f2);
}

fclose(f1);
fclose(f2);

printf("file copied");

delay(5000);
}
