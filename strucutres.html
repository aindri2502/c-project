#include <stdio.h>
#include <conio.h>
#include <dos.h>

struct employee
{
int id;
char name[20];
char designation[20];
};

void main()
{
struct employee ep[5];
int i;
clrscr();

for(i=0;i<5;i++)
{
printf("\nPlease give your employee id= ");
scanf("%d",&ep[i].id);

printf("name= ");
scanf("%s",ep[i].name);

printf("designation= ");
scanf("%s",ep[i].designation);
}

for(i=0;i<5;i++)
{
printf("%d %s %s\n",ep[i].id,ep[i].name,ep[i].designation);
}

delay(5000);
}
